                <?php
                if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['test_usuario'])) {
                    try {
                        $idUsuario = (int)($_POST['id_usuario'] ?? 0);
                        
                        if ($idUsuario <= 0) {
                            echo "<div class='box error'><strong>✗ Error:</strong> ID de usuario inválido.</div>";
                        } else {
                            $db = (new Conexion())->conectar();
                            $usuarioModel = new Usuario($db);
                            $usuario = $usuarioModel->getUsuarioById($idUsuario);
                            
                            if (!$usuario) {
                                echo "<div class='box error'><strong>✗ Error:</strong> Usuario no encontrado.</div>";
                            } elseif (empty($usuario['correo'])) {
                                echo "<div class='box warning'><strong>⚠️ Advertencia:</strong> El usuario no tiene correo registrado.</div>";
                            } else {
                                $mailer = new Mailer();
                                $subject = 'Prueba desde BibliotecKJ - ' . date('Y-m-d H:i:s');
                                $content = "Estimado(a) " . htmlspecialchars($usuario['nombre']) . ",<br><br>" .
                                          "Este es un correo de prueba desde el sistema BibliotecKJ.<br>" .
                                          "Si lo recibiste, tu correo está correctamente registrado.";
                                
                                $resultado = $mailer->send($usuario['correo'], $subject, $content);
                                
                                if ($resultado) {
                                    echo "<div class='box success'><strong>✓ Éxito:</strong> Correo enviado a " . htmlspecialchars($usuario['correo']) . "</div>";
                                } else {
                                    echo "<div class='box error'><strong>✗ Error:</strong> No se pudo enviar el correo al usuario.</div>";
                                }
                            }
                        }
                    } catch (Exception $e) {
                        echo "<div class='box error'><strong>✗ Excepción:</strong> " . htmlspecialchars($e->getMessage()) . "</div>";
                    }
